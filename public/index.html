// Hàm mở Modal và hiển thị các ngày mưa đã lưu trong Database async function openRainModal() { document.getElementById('rain-modal').classList.remove('hidden'); const container = document.getElementById('rain-grouped-container'); container.innerHTML = `
Đang tải lịch sử các đợt mưa từ Database...
`; try { const res = await fetch('/api/rain-history'); const data = await res.json(); if (!data.history || data.history.length === 0) { container.innerHTML = `
☀️
Chưa có đợt mưa thực tế nào được ghi nhận!

Hệ thống đã lọc bỏ các vệt ẩm mô hình nhỏ hơn 0.5 mm.

`; return; } // Gom nhóm các đợt mưa theo ngày const groups = {}; data.history.forEach(r => { let d = r.rain_date; if (!d && r.start_time) { const parts = r.start_time.split(' '); if (parts.length > 1) d = parts[1]; } if (!d) d = 'Không rõ ngày'; if (!groups[d]) groups[d] = []; groups[d].push(r); }); const nowVN = new Date(new Date().toLocaleString("en-US", { timeZone: "Asia/Ho_Chi_Minh" })); const pad = (n) => String(n).padStart(2, '0'); const todayStr = `${pad(nowVN.getDate())}/${pad(nowVN.getMonth() + 1)}/${nowVN.getFullYear()}`; // Sắp xếp ngày mới nhất lên trên cùng const sortedDates = Object.keys(groups).sort((a, b) => { const [d1, m1, y1] = a.split('/').map(Number); const [d2, m2, y2] = b.split('/').map(Number); return new Date(y2, m2 - 1, d2) - new Date(y1, m1 - 1, d1); }); container.innerHTML = sortedDates.map(dateKey => { const dayEvents = groups[dateKey]; dayEvents.sort((a, b) => (a.episode_no || 1) - (b.episode_no || 1)); const totalMmDay = dayEvents.reduce((sum, item) => sum + (parseFloat(item.total_mm) || 0), 0); const isToday = (dateKey === todayStr); return `
🌧️ NGÀY: ${dateKey} ${isToday ? 'HÔM NAY' : ''}
Tổng: ${dayEvents.length} đợt mưa | Lượng mưa cả ngày: ${totalMmDay.toFixed(2)} mm
${dayEvents.map(ev => { const isOngoing = ev.end_time.includes('Đang'); const endDisplay = isOngoing ? 'Đang mưa' : ev.end_time.split(' ')[0]; return ` `; }).join('')}
Tên Đợt Mưa lúc (Bắt đầu) Tạnh lúc (Kết thúc) Thời lượng Khoảng cách với đợt trước Đỉnh mưa / Tổng
Đợt ${ev.episode_no || 1} ${ev.start_time.split(' ')[0]} ${endDisplay} ⏱️ ${ev.duration_min} phút ${ev.episode_no === 1 ? 'Đợt đầu tiên trong ngày' : `💤 ${ev.gap_desc}`} ${ev.peak_mm} mm / ${ev.total_mm} mm
`; }).join(''); } catch (err) { console.error("Lỗi tải lịch sử mưa:", err); } }
